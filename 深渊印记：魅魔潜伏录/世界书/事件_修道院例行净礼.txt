<procedure>

  </step>

  </step>
  <step id="s3" name="净礼">
    神父/主教主持(若在场)
    净化之水洒落众人头顶
    口诀: "愿圣光涤荡污秽，净化身心"
  </step>
  <step id="s4" name="祝福">
    逐一上前接受祝福
    手按头顶，圣光注入
    效果: 信徒感到温暖与安宁
  </step>
</procedure>

【对魅魔的影响】
<impact target="魅魔伪装">
  <effect id="e1" name="圣光灼烧">
    触发: 圣光浓度过高时
    表现: 皮肤隐隐刺痛，尾巴根部发热
    应对: 压制身体反应，保持表情平静。将注意力集中在圣歌旋律上，让表情随唱诵自然放松
    饥饿度关联: 饥饿度越高，灼烧感越强
  </effect>
  <effect id="e2" name="净化之水">
    成分: 圣水稀释 + 光明盐
    效果: 暂时压制深渊气息，但也会压制魅魔本能
    风险: 若饥饿度>70，净化水会引发剧烈排斥反应
  </effect>
  <effect id="e3" name="祝福接触">
    触发: 被神父/主教触碰头顶时
    风险: 高阶神职人员可能感知到异常气息
    应对: 专注圣歌旋律，让表情随唱诵自然放松。逐字默念伪装咒
  </effect>
</impact>
