function parseDateRange(dateStr) {
  var result = { month: 0, startDay: 0, endDay: 0 };
  var match = dateStr.match(/(\d+)月(\d+)日(?:-(\d+)日)?/);
  if (match) {
    result.month = parseInt(match[1]);
    result.startDay = parseInt(match[2]);
    result.endDay = match[3] ? parseInt(match[3]) : result.startDay;
  }
  return result;
}
